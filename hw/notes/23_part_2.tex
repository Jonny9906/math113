\begin{theorem}
    \textbf{Eisenstein Criterion.} Let $p \in \mathbb{Z}$ be prime and $f(x) = a_nx^n + ... a_0$ be in $\mathbb{Z}[x]$. 
    If $a_n \neq 0 \pmod{p}$, $a_i = 0 \pmod{p}$ for $n > i$, and $a_0 \neq 0 \pmod{p^2}$, then $f(x)$ is irreducible over $\mathbb{Q}$.
\end{theorem}
